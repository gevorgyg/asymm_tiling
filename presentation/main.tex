\begin{frame}{Where Does Mixed Precision Arise?}

  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \textbf{Stored at lower precision}
      \begin{itemize}
        \item Low-bit ML inference (fp16, bfloat16, int8)
        \item Quantized weight matrices
        \item Randomized sketching in linear algebra
      \end{itemize}

      \bigskip
      \textbf{Generated on the fly (PRNG)}
      \begin{itemize}
        \item $B$ is a pseudorandom matrix
        \item Hardware FIFO streams random values at fixed rate
      \end{itemize}
    \end{column}

    \begin{column}{0.49\textwidth}
      \bigskip
      \bigskip
      \bigskip
      \textbf{Question:} Standard matmul (DGEMM) is optimized for $\rho = 1$.
      \medskip

      Can we do better by changing the \textbf{shape} of the computation tiles?
    \end{column}
  \end{columns}
\end{frame}

% ── T1: cost model ─────────────────────────────────────────────────────────────
\begin{frame}{The Memory Traffic Model}

  For one tile of $C$ (size $T_M \times T_N$, fast memory $M = T_M \cdot T_N$):

  \bigskip
  \begin{columns}[c]
    \begin{column}{0.54\textwidth}
      \renewcommand{\arraystretch}{1.6}
      \begin{tabular}{lll}
        \toprule
        Matrix & Reloads/elem & Cost \\
        \midrule
        $A$ (expensive) & $n / T_N$ & $\propto \dfrac{1}{T_N}$ \\[6pt]
        $B$ (cheap, $\rho$) & $m / T_M$ & $\propto \dfrac{\rho}{T_M}$ \\
        \bottomrule
      \end{tabular}

      \bigskip
      Minimize $\dfrac{1}{T_N} + \dfrac{\rho}{T_M}$ subject to $T_M \cdot T_N = M$.
    \end{column}

    \begin{column}{0.42\textwidth}
      \begin{block}{Optimal tile shape}
        \[
          \frac{T_N}{T_M} = \frac{1}{\rho}
        \]
        Wide tile when $B$ is cheap ($\rho < 1$).
      \end{block}

      \smallskip
      Examples:
      \begin{itemize}
        \item $\rho = 1$ (equal precision): $T_N = T_M$ \; \textbf{(square)}
        \item $\rho = 1/4$ (fp16 vs fp64): $T_N = 4\,T_M$
        \item $\rho \to 0$ (PRNG FIFO): widest feasible tile
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ── T1b: derivation ───────────────────────────────────────────────────────────
\begin{frame}{Deriving the Optimal Tile Shape}

  Parametrize by $z = T_M$ with constraint $T_M \cdot T_N = M$, so $T_N = M/z$:
  \[
    f(z) \;=\; mnk\!\left(\frac{A_p \cdot z}{M} + \frac{B_p}{z}\right) + mn\cdot C_p
  \]

  \smallskip
  Differentiate and set to zero:
  \begin{align*}
    f'(z) \;=\; mnk\!\left(\frac{A_p}{M} - \frac{B_p}{z^2}\right) &= 0
    \\[4pt]
    \Rightarrow\quad z^2 &= \frac{B_p}{A_p}\cdot M \;=\; \rho M
  \end{align*}

  \smallskip
  Substituting back ($z = T_M$, $M = T_M \cdot T_N$):
  \[
    T_M^2 \;=\; \rho\,T_M T_N
    \quad\Rightarrow\quad
    T_M \;=\; \rho\,T_N
    \quad\Rightarrow\quad
    \boxed{\dfrac{T_N}{T_M} \;=\; \dfrac{1}{\rho}}
  \]

\end{frame}

% ── T2: speedup formula ────────────────────────────────────────────────────────
\begin{frame}{Analytical Speedup over Square Tiles}

  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      At the optimal tile the two traffic terms balance:
      \[
        \underbrace{\frac{A_P}{T_N}}_{\text{A traffic}} = \underbrace{\frac{B_P}{T_M}}_{\text{B traffic}}
      \]
      Substituting $T_N^\star = \sqrt{M/\rho}$, $T_M^\star = \sqrt{M\rho}$:
      \[
        \mathrm{speedup} = \frac{\text{cost(square)}}{\text{cost(optimal)}}
                         = \frac{1 + \rho}{2\sqrt{\rho}}
      \]
    \end{column}

    \begin{column}{0.46\textwidth}
      \begin{block}{Predicted speedups}
        \renewcommand{\arraystretch}{1.35}
        \begin{tabular}{ccc}
          \toprule
          $\rho$ & $B$ precision & speedup \\
          \midrule
          $1$       & fp64 (8\,B) & $1.00\times$ \\
          $1/2$     & fp32 (4\,B) & $1.06\times$ \\
          $1/4$     & fp16 (2\,B) & $1.25\times$ \\
          $1/8$     & int8 (1\,B) & $1.59\times$ \\
          $\to 0$   & PRNG FIFO   & $\to \infty$ \\
          \bottomrule
        \end{tabular}
      \end{block}
      \smallskip
      {\footnotesize Speedup grows monotonically as $B$ becomes cheaper.}
    \end{column}
  \end{columns}
\end{frame}

% ── T2b: speedup derivation ───────────────────────────────────────────────────
\begin{frame}{Where the Speedup Formula Comes From}

  Traffic (ignoring $C$): $\;f = mnk\!\left(\dfrac{A_p}{T_N} + \dfrac{B_p}{T_M}\right)$

  \medskip
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \textbf{Square tile} \quad $T_M = T_N = \sqrt{M}$:
      \[
        f_{\text{sq}} = \frac{mnk\,(A_p + B_p)}{\sqrt{M}}
      \]
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{Optimal tile} \quad $T_M^\star = \sqrt{\rho M}$, $\;T_N^\star = \sqrt{M/\rho}$:
      \begin{align*}
        f_{\text{opt}} &= \frac{mnk}{\sqrt{M}}\!\left(A_p\sqrt{\rho} + \frac{B_p}{\sqrt{\rho}}\right)
      \end{align*}
      With $B_p = \rho A_p$: \quad $A_p\sqrt{\rho} + \dfrac{\rho A_p}{\sqrt{\rho}} = 2A_p\sqrt{\rho}$
      \[
        f_{\text{opt}} = \frac{2\,mnk\,A_p\sqrt{\rho}}{\sqrt{M}}
      \]
    \end{column}
  \end{columns}

  \medskip
  \[
    \mathrm{speedup}
      = \frac{f_{\text{sq}}}{f_{\text{opt}}}
      = \frac{A_p + B_p}{2A_p\sqrt{\rho}}
      = \frac{A_p(1+\rho)}{2A_p\sqrt{\rho}}
      = \boxed{\dfrac{1+\rho}{2\sqrt{\rho}}}
  \]

  {\footnotesize \textit{Note:} at the optimum $A_p\sqrt{\rho} = B_p/\sqrt{\rho}$ — both input terms carry equal traffic.}

\end{frame}

\section{The Simulator}

\begin{frame}{Cycle-Accurate Cache Simulator}

  \begin{columns}[T]

    \begin{column}{0.48\textwidth}
      \textbf{How it works}\\[1pt]
      {\small
      \begin{itemize}\setlength\itemsep{0pt}
        \item An \textbf{instruction stream generator} emits a \texttt{.matv} trace
              for a given tile shape and ordering
        \item A \textbf{cycle-accurate interpreter} replays the trace through
              L1 + L2 + DRAM
        \item Output: total cycles, per-level traffic, hit/miss counts
      \end{itemize}}

      \smallskip
      \textbf{Instruction set}\\[1pt]
      {\small
      \begin{itemize}\setlength\itemsep{0pt}
        \item \texttt{ltea} — load a register tile from memory
        \item \texttt{tmov} — consume values from the PRNG FIFO
        \item \texttt{prefetch} — issue a prefetch hint
        \item \texttt{tmulac} — register-tile multiply-accumulate
      \end{itemize}}
    \end{column}

    \begin{column}{0.48\textwidth}
      \textbf{Two modes for $B$}\\[1pt]
      {\small
      \begin{itemize}\setlength\itemsep{0pt}
        \item \textbf{From memory} — $B$ cached normally like $A$ and $C$
        \item \textbf{PRNG FIFO} — hardware stream device generates values
              on the fly; zero cache footprint ($\rho \to 0$)
      \end{itemize}}

      \smallskip
      \textbf{Configurable parameters}\\[1pt]
      {\small
      \begin{itemize}\setlength\itemsep{0pt}
        \item L1 / L2: size, line size, associativity, replacement policy
        \item Element widths $A_p$, $B_p$ (bytes per element)
        \item Tile dimensions $T_M$, $T_N$, $T_K$
        \item Register tile $\mathrm{REG}_M \times \mathrm{REG}_N \times \mathrm{REG}_K$
      \end{itemize}}
    \end{column}

  \end{columns}

\end{frame}

\begin{frame}{Let's Test the Theory}

  We have closed-form predictions for L1 traffic:
  \[
    \text{reads} = mnk\!\left(\frac{A_p}{T_N} + \frac{B_p}{T_M}\right) + mn\cdot C_p
    \qquad
    \text{writes} = mn \cdot C_p
  \]
  with the optimum at $T_N / T_M = 1/\rho$.

  \bigskip
  We run the simulator over a sweep of tile aspect ratios and overlay
  the measured traffic against the closed-form curves.

\end{frame}

% ── AM-GM balance: derivation ────────────────────────────────────────────────
\begin{frame}{Why A and B Traffic Must Be Equal at the Optimum}

  Separate the two \emph{input} terms from the reads formula:
  \[
    \underbrace{\frac{mnk\,A_p}{T_N}}_{\text{A traffic}} \;+\; \underbrace{\frac{mnk\,B_p}{T_M}}_{\text{B traffic}} \;+\; mn C_p
  \]

  \medskip
  Their ratio (word model):
  \[
    \frac{\text{B traffic}}{\text{A traffic}}
      \;=\; \frac{B_p / T_M}{A_p / T_N}
      \;=\; \frac{B_p}{A_p}\cdot\frac{T_N}{T_M}
      \;=\; \rho \cdot \frac{T_N}{T_M}
  \]

  \medskip
  Plug in the optimal ratio $T_N/T_M = 1/\rho$:
  \[
    \frac{\text{B traffic}}{\text{A traffic}}\bigg|_{\text{opt}}
      \;=\; \rho \cdot \frac{1}{\rho} \;=\; \boxed{1}
  \]

  \smallskip
  {\small The AM-GM step in the derivation is exactly this balance condition —
  the minimum is achieved when the two input terms carry \emph{equal} traffic.}

\end{frame}

% ── Model validity: assumption reminder ───────────────────────────────────────
\begin{frame}{When Does the Paper's Model Hold?}
  \begin{block}{Paper's assumption}
    The C block occupies essentially \textbf{all} of fast memory $M$.
    Subrows of $A$ and subcolumns of $B$ stream \emph{through} — they are used
    once and evicted; they do not compete with $C$ for residency.
  \end{block}

  \medskip
  This breaks when the C tile is \textbf{too large}:

  \begin{itemize}
    \item A strip has $T_M$ rows of $k$ elements — it needs some L1 space too
    \item If the C tile already fills most of L1, the streaming lines evict C
          before the block is fully accumulated → extra re-fetches → model underpredicts traffic
  \end{itemize}

  \medskip
  \textbf{Question:} how large can the C tile actually be before the model breaks?

  \smallskip
  {\small We fix the aspect ratio at the predicted optimum ($T_N/T_M = 4$, $\rho=\tfrac{1}{4}$)
  and grow the C tile, measuring $\text{excess} = \text{measured} / \text{predicted}$.
  Excess $= 1$ means the model holds exactly.}

\end{frame}
