\begin{frame}{Asymmetric Precision $\Rightarrow$ Asymmetric Tiles}
  \begin{columns}[c]

    \begin{column}{0.50\textwidth}
      \\
      \medskip

      $\#A \; Bytes = T_M \cdot k \cdot A_p$\\
      $\#B \; Bytes = T_N \cdot k \cdot B_p$\\

      \medskip

      $\Rightarrow T = \frac{MN}{T_MT_N}\!\left(T_M \cdot k \cdot A_p + T_N \cdot k \cdot B_p\right)$\\

      \medskip
      
      \[
          \boxed{T = MNK\!\left(\frac{A_P}{T_N} + \frac{B_P}{T_M}\right)}
      \]

      \medskip

      Find best ratio:
      \[
        \;\;\Rightarrow\;\;
        \frac{T_N^*}{T_M^*} = \frac{A_P}{B_P} = \frac{1}{\rho}
      \]
    \end{column}

    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.90]
        % Both tiles share the same TM height = 1.2
        \def\tmH{1.2}

\section{From Traffic to Time}
% ==============================================================================

\begin{frame}{Traffic $\neq$ Time}


\begin{center}
  The paper minimizes \emph{bytes transferred}:\\
  \medskip
  \[
    T_{\text{traffic}} = MNK\!\left(\frac{A_P}{T_N} + \frac{B_P}{T_M}\right)
  \]

  \bigskip
  \textbf{But cycles $\neq$ bytes.}

  \medskip
  We need a \emph{cycle-accurate} model.
\end{center}

\end{frame}


\begin{frame}{Enter the PRNG FIFO}

  \begin{columns}[c]

    \begin{column}{0.50\textwidth}
      \begin{block}{Enter the PRNG FIFO}
        B elements are generated \emph{on-chip} by a hardware random
        number generator, streamed directly into the compute unit.\\[6pt]
        \textbf{B never touches the cache.}\\
        \textbf{B has no memory address.}
      \end{block}
    \end{column}

    \begin{column}{0.46\textwidth}
      The $B_P / T_M$ traffic term \textbf{vanishes}.\\[8pt]
      Instead, B introduces a new cost: on-chip \emph{generation time}.
      \[
        \text{generation cost} = g_c \;\text{cycles / element}
      \]
    \end{column}

  \end{columns}

\end{frame}

\section{Building the Model}
\begin{frame}{Naive Cycle Model: Two Cost Terms}


    \begin{center}
      Same reuse argument as traffic, but in \textbf{cycles}:

      \bigskip
      Naively assuming the two costs \textbf{add}:
      \[
        \frac{T}{MNK} = \frac{C_A}{T_N} + \frac{g_c}{T_M}
      \]
    \end{center}
\end{frame}

% ── Slide 12: FIFO is async → max not sum ────────────────────────────────────
\begin{frame}{FIFO is Async: Two Costs in Parallel}


    \begin{center}
      The FIFO generates B \textbf{in the background}\\
      while the core loads A from L1.

      \medskip
      The two costs \textbf{overlap} --- runtime is\\
      determined by whichever finishes last:

      \[
        \frac{T}{MNK} = \max\!\left\{\frac{C_A}{T_N},\;\frac{g_c}{T_M}\right\}
      \]

      \medskip
      \textbf{Two regimes:}\\
      \medskip
      $g_c/T_M < C_A/T_N$:\; A-load bound\\
      $g_c/T_M > C_A/T_N$:\; B-gen bound\\
    \end{center}
\end{frame}

% ── Slide 13: Replace C_A/TN with measured α(TM,TN) ─────────────────────────
\begin{frame}{Replacing $C_A/T_N$ with Measured $\alpha(T_M,T_N)$}
    \begin{center}
      The $C_A/T_N$ term assumes:\\
    \medskip
        $C_A$ = fixed cost per A element from L1\\
        $T_N$-fold reuse divides it out cleanly\\

      \medskip
      \textbf{Problem:} actual cycles/MAC depends on
      tile shape through cache residency, line
      utilization, and access patterns ---
      not a simple constant divided by $T_N$.

      \medskip
      \textbf{Solution:} 
      \[
          \boxed{\frac{T}{MNK} = \max\!\left\{
          \alpha(T_M,T_N),\;\frac{g_c}{T_M}
          \right\}}
      \]

    \end{center}
\end{frame}

% ── Slide 14: Measuring α by setting gc=0 ────────────────────────────────────
\begin{frame}{Isolating $\alpha$: Set $g_c = 0$}
    \begin{center}
      \textbf{Key trick:} set $g_c = 0$

      \medskip
      The model collapses:
      \[
        \frac{T}{MNK} = \max\!\left\{\alpha(T_M,T_N),\;\frac{0}{T_M}\right\}
        = \alpha(T_M,T_N)
      \]
    \end{center}
\end{frame}
