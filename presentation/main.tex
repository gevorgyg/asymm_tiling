\section{Square Tiles}
% ==============================================================================

\begin{frame}{Standard Assumption: Square Tiles}
  \begin{columns}[c]

    \begin{column}{0.50\textwidth}
      Each A element is reused $T_N$ times (once per C column).\\
      Each B element is reused $T_M$ times (once per C row).

      \bigskip
      \[
        T = MNK\!\left(\frac{A_P}{T_N} + \frac{B_P}{T_M}\right)
      \]

      \bigskip
      For equal precision ($A_P = B_P$), minimize $\tfrac{1}{T_M}+\tfrac{1}{T_N}$
      subject to $T_M T_N \le S$:

      \medskip
      \[
        \boxed{T_M^{*} = T_N^{*} \quad (\text{square tiles})}
      \]
    \end{column}

    \begin{column}{0.44\textwidth}

\section{Asymmetric Precision}
% ==============================================================================

\begin{frame}{Asymmetric Precision $\Rightarrow$ Asymmetric Tiles}
  \begin{columns}[c]

    \begin{column}{0.50\textwidth}
      Let $\rho = B_P / A_P$ \quad (bytes per element).\\[4pt]
      When $B_P < A_P$: B is cheaper to load $\Rightarrow$ use wider tiles.

      \medskip
      \[
        T = MNK\!\left(\frac{A_P}{T_N} + \frac{B_P}{T_M}\right)
      \]

      Balance A and B traffic:
      \[
        \frac{A_P}{T_N} = \frac{B_P}{T_M}
        \;\;\Rightarrow\;\;
        \boxed{\frac{T_N^*}{T_M^*} = \frac{A_P}{B_P} = \frac{1}{\rho}}
      \]

      \medskip
      \textbf{Example:} fp32 $\times$ fp4 ($\rho = \tfrac{1}{8}$):
      \[
        T_N^* = 8\, T_M^*
      \]
    \end{column}

    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        % Symmetric tile (rho=1)
        \node[font=\small, gray] at (1.1, 3.55) {$\rho=1$\ (fp32 $\times$ fp32)};

\section{From Traffic to Time}
% ==============================================================================

\begin{frame}{Traffic $\neq$ Time}

  \begin{columns}[c]

    \begin{column}{0.46\textwidth}
      The paper minimizes \emph{bytes transferred}:
      \[
        T_{\text{traffic}} = MNK\!\left(\frac{A_P}{T_N} + \frac{B_P}{T_M}\right)
      \]
      \textbf{But cycles $\neq$ bytes.}\\[6pt]
      Cache latency, bandwidth limits, and prefetch behavior all
      break the proportionality between traffic and runtime.

      \medskip
      We need a \emph{cycle-accurate} model.
    \end{column}

    \begin{column}{0.50\textwidth}
      \begin{block}{Enter the PRNG FIFO}
        B elements are generated \emph{on-chip} by a hardware random
        number generator, streamed directly into the compute unit.\\[6pt]
        \textbf{B never touches the cache.}\\
        \textbf{B has no memory address.}
      \end{block}

      \medskip
      The $B_P / T_M$ traffic term \textbf{vanishes}.\\[6pt]
      Instead, B introduces a new cost: on-chip \emph{generation time}.\\[6pt]
      \[
        \text{generation cost} = g_c \;\text{cycles / element}
      \]
      Our model must account for $g_c$, not $B_P$.
    \end{column}

  \end{columns}

\end{frame}

% ── Slide 7: which loop order fits the FIFO? ─────────────────────────────────
\begin{frame}{Which Loop Order Fits the FIFO?}

  \begin{columns}[c]

    % ── Left: comparison table ────────────────────────────────────────────────
    \begin{column}{0.55\textwidth}
      B elements emerge from the PRNG in a \textbf{fixed order}\\
      --- you cannot skip or replay elements.

      \medskip
      Three loop structures are possible:

      \bigskip
      \renewcommand{\arraystretch}{1.55}
      \begin{tabular}{@{}p{2.4cm}p{1.6cm}l@{}}
        \toprule
        Mode & In regs & B cost / elem \\
        \midrule
        C-stat row-maj  & $C_{ij}$ tile & $g_c \times T_N$ (waste) \\[2pt]
        C-stat col-maj  & $C_{ij}$ tile & $g_c$ \\[2pt]
        B-stationary    & B block (outer) & $\boldsymbol{g_c / T_M}$ \\

% ── Slide 19: Why the model breaks — the gen-bound regime ─────────────────────
\begin{frame}{Why $T_N^*$ Fails at High $g_c$: Gen-Bound Regime}

  \begin{columns}[c]

    \begin{column}{0.44\textwidth}
      When $g_c / T_M \gg \alpha(T_M, T_N)$ for all $T_N$:

      \[
        \max\!\left\{\alpha, \frac{g_c}{T_M}\right\}
        \approx \frac{g_c}{T_M}
        \quad (\text{$T_N$-independent})
      \]

      The model predicts the same cost for every $T_N$
      --- it cannot distinguish between them.

      \medskip
      \textbf{Consequence:}
      \begin{itemize}
        \item $T_M^*$ always correct ($g_c/T_M$ is TM-dependent)
        \item $T_N^*$ chosen arbitrarily by the model
        \item Empirically $T_N{=}16$ is best in
          gen-bound regime (pipeline effects)
      \end{itemize}
    \end{column}
